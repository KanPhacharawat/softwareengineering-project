# API demonstration against local release (server on :3100, cpms_e2e seeded)

$ curl -s -i http://localhost:3100/api/teams | sed -n '1p;$p'
HTTP/1.1 401 Unauthorized
{"error":{"code":"UNAUTHENTICATED","message":"Please log in"}}

$ curl -s -c cookies-s1 -o /dev/null -w '%{http_code}\n' -H 'Content-Type: application/json' -d '{"email":"s1@cpms.test","password":"Passw0rd!"}' http://localhost:3100/api/auth/login
200


$ curl -s -b cookies-s1 http://localhost:3100/api/auth/me
{"user":{"id":"210a748a-bfaa-4c0d-a424-2ea71a86ce10","email":"s1@cpms.test","name":"Chai Student","role":"STUDENT"}}
